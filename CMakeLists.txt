function(slx_target name)
  target_include_directories(${name} PUBLIC "${CMAKE_SOURCE_DIR}/sdk/include")
  target_compile_definitions(${name} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(${name} SYSTEM PRIVATE "${SLX_DEPS}/include")
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.")
  endif()
  if(MINGW)
    target_link_options(${name} PRIVATE -static-libgcc -static-libstdc++)
  endif()
endfunction()
function(slx_copy_core name)
  if(NOT SLX_BUILD_PRIVATE_CORE)
    add_custom_command(TARGET ${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${SLX_CORE_BINARY}" "$<TARGET_FILE_DIR:${name}>")
    add_custom_command(TARGET ${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_SOURCE_DIR}/core/windows-x64/slx-eebus.exe" "$<TARGET_FILE_DIR:${name}>")
  endif()
endfunction()
if(SLX_BUILD_PRIVATE_CORE)
  find_program(SLX_GO_EXECUTABLE go.exe HINTS "${CMAKE_SOURCE_DIR}/build/tools/go/bin" REQUIRED)
  file(GLOB SLX_SUNSPEC_MODELS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/private/eebus/sunspec_models/*.json")
  file(GLOB SLX_SERVICE_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/private/eebus/*.go")
  file(GLOB_RECURSE SLX_PATCHED_EEBUS_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/private/eebus/third_party/eebus-go/*.go")
  add_custom_command(OUTPUT "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/slx-eebus.exe"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}"
    COMMAND ${CMAKE_COMMAND} -E env "GOPATH=${CMAKE_SOURCE_DIR}/build/go-cache" GOTOOLCHAIN=local CGO_ENABLED=0
      "${SLX_GO_EXECUTABLE}" build -trimpath -buildvcs=false "-ldflags=-s -w" -o "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/slx-eebus.exe" .
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}/private/eebus"
    DEPENDS ${SLX_SERVICE_SOURCES} ${SLX_PATCHED_EEBUS_SOURCES} ${SLX_SUNSPEC_MODELS} private/eebus/go.mod private/eebus/go.sum
    VERBATIM)
  add_custom_target(slx-eebus-bridge ALL DEPENDS "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/slx-eebus.exe")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(NNG_TESTS OFF CACHE BOOL "" FORCE)
  set(NNG_TOOLS OFF CACHE BOOL "" FORCE)
  set(NNG_ENABLE_TLS OFF CACHE BOOL "" FORCE)
  set(NNG_ENABLE_HTTP OFF CACHE BOOL "" FORCE)
  set(NNG_ENABLE_COMPAT OFF CACHE BOOL "" FORCE)
  add_subdirectory("${SLX_DEPS}/nng" "${CMAKE_BINARY_DIR}/nng" EXCLUDE_FROM_ALL)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(nng PRIVATE "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.")
  endif()
  add_library(slx-solarpilot-core SHARED private/core/core.cpp private/core/model.cpp private/core/link.cpp private/core/eebus.cpp)
  add_dependencies(slx-solarpilot-core slx-eebus-bridge)
  slx_target(slx-solarpilot-core)
  target_compile_definitions(slx-solarpilot-core PRIVATE SLX_SP_BUILD_CORE)
  target_link_libraries(slx-solarpilot-core PRIVATE nng)
  set_target_properties(slx-solarpilot-core PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  if(MINGW)
    target_link_options(slx-solarpilot-core PRIVATE -Wl,--exclude-all-symbols)
  endif()
else()
  set(SLX_CORE_BINARY "${CMAKE_SOURCE_DIR}/core/windows-x64/slx-solarpilot-core.dll" CACHE FILEPATH "Supplied ABI 1 library")
  add_library(slx-solarpilot-core SHARED IMPORTED GLOBAL)
  set_target_properties(slx-solarpilot-core PROPERTIES IMPORTED_LOCATION "${SLX_CORE_BINARY}" IMPORTED_IMPLIB "${CMAKE_SOURCE_DIR}/core/windows-x64/libslx-solarpilot-core.dll.a")
endif()
